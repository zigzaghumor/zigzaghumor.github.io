<script>
  (function () {
    var saved = null;
    try { saved = window.localStorage.getItem('site_lang'); } catch (e) {}
    var prefersZh = window.location.pathname.indexOf('/zh') === 0 || saved === 'zh-CN';
    if (!prefersZh) return;

    document.documentElement.lang = 'zh-CN';
    document.title = '404 页面未找到 | Zhihang Zhao | 影石 Insta360';
    var heading = document.getElementById('error-heading');
    var primary = document.getElementById('error-lead-primary');
    var secondary = document.getElementById('error-lead-secondary');
    var actions = document.querySelector('.error-actions');
    var btnZh = document.getElementById('error-btn-zh');
    var btnEn = document.getElementById('error-btn-en');
    if (heading) heading.textContent = '页面未找到';
    if (primary) {
      primary.textContent = '未找到该页面，链接可能已失效或地址有误。';
      primary.lang = 'zh-CN';
    }
    if (secondary) {
      secondary.textContent = 'The page you are looking for does not exist or may have moved.';
      secondary.lang = 'en';
    }
    if (actions && btnZh && btnEn) {
      actions.insertBefore(btnZh, btnEn);
    }
  })();
</script>
